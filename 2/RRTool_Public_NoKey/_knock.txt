{
    uint64_t state, model, wrap, val;
    int v;

    if (!character) return 0;
    state = rr_u64(character + RR_OFF_CHAR_STATE);
    if (!state || !PL(state) || !u_is(state, "BattleCharacterStateModel")) return 0;

    model = rr_u64(state + RR_OFF_CHAR_KNOCK_MODEL);
    if (!model || !PL(model)) return 0;

    wrap = rr_u64(model + RR_OFF_KNOCK_PHASE);
    if (!wrap || !PL(wrap)) return 0;

    if (!rr_read(wrap + RR_OFF_REF_VALUE, &val, sizeof(val))) return 0;
    if (!val || !PL(val) || !u_is(val, "KnockoutPhaseDescription")) return 0;

    v = rr_phase_verdict(val);
    return v > 0;
}
