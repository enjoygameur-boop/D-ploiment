----- rr_aim_step -----
rr_aim_step(RR_AIM_STEER* state, float dx, float dy, float fov, int height,
                 int smooth, int cal, unsigned elapsed_ms, int firing, int* mx, int* my) {
    *mx = *my = 0;
    if (!isfinite(dx) || !isfinite(dy) || !(fov > 10.0f && fov < 170.0f) || height < 16) {
        *state = (RR_AIM_STEER){0};
        return;
    }
    /* Project to the equivalent error at 1080p / 60 degrees. The same
     * angular error should produce the same correction at a different zoom. */
    float scale = (1080.0f / height) * (tanf(fov * 3.14159265f / 360.0f) / 0.57735027f);
    /* Convert the normalized pixel error to mouse counts using cal. This
     * ratio depends on the game sensitivity; --aimtest measures it for the current
     * setup. A value of 100 assumes one count per reference pixel.
     */
    if (!(cal > 0)) cal = 100;
    float calib = (float)cal / 100.0f;
    scale *= calib;
    /* Scale the per-step limit with calibration so it represents a similar
     * angular correction at different sensitivities.
     */
    float limit = fmaxf(8.0f, 32.0f * calib);
    float dt = (float)(elapsed_ms > 24 ? 24 : elapsed_ms);
    /* Ease the correction in as the error shrinks, so the aim settles onto the
     * point instead of arriving at full speed. */
    float ex = fabsf(dx) > fabsf(dy) ? fabsf(dx) : fabsf(dy);
    float err_deg = atanf((ex * 2.0f / height) / tanf(fov * 3.14159265f / 360.0f)) * 57.295779513f;
    float slow = err_deg < 8.0f ? 1.0f + 2.0f * (1.0f - err_deg / 8.0f) : 1.0f;
    float response = aim_response_ms(smooth) * slow;
    float gain = 1.0f - expf(-dt / response);
    float vertical_gain = firing ? 1.0f - expf(-dt / (response * 0.8f)) : gain;
    *mx = aim_axis(dx * scale, gain, limit, &state->carry_x);
    *my = aim_axis(dy * scale, vertical_gain, limit, &state->carry_y);
}
----- aim_axis -----
aim_axis(float error, float gain, float limit, float* carry) {
    if (fabsf(error) < 0.05f) {
        *carry = 0.0f;
        return 0;
    }
    if (error * *carry < 0.0f) *carry = 0.0f;
    /* A large correction must not be rate-limited by the smoothing limit, or
     * the first frames after a flick stay behind a fast target. */
    float need = fabsf(error) * 0.5f;
    if (need > limit) limit = need < 160.0f ? need : 160.0f;
    float step = error * gain + *carry;
    /* Never stall short of the point: one count while the error is a pixel. */
    if (fabsf(step) < 1.0f && fabsf(error) >= 1.0f) step = error > 0.0f ? 1.0f : -1.0f;
    step = fmaxf(-limit, fminf(limit, step));
    int counts = (int)step;
    *carry = step - counts;
    return counts;
}

----- rr_aim_motion_point -----
rr_aim_motion_point(RR_V3 point, RR_V3 position, RR_V3 previous,
                        unsigned sample_ms, unsigned age_ms, int smooth, RR_V3* out) {
    if (sample_ms < 8 || sample_ms > 120 || age_ms > 80) return 0;
    float vx = (position.x - previous.x) * 1000.0f / sample_ms;
    float vz = (position.z - previous.z) * 1000.0f / sample_ms;
    float speed = hypotf(vx, vz);
    if (!(speed >= 0.2f && speed <= 20.0f)) return 0;

    /* Lead through the whole smoothing delay, not just part of it, plus the
     * age of the snapshot the point came from. At equilibrium that cancels the
     * lag of an exponential follow, so a constant-velocity target settles on
     * the crosshair instead of trailing behind it. The cap bounds the
     * extrapolation error when a target changes direction mid-lead. */
    float seconds = (aim_response_ms(smooth) + (float)age_ms) * 0.001f;
    seconds = fminf(seconds, 0.20f);
    *out = (RR_V3){ point.x + vx * seconds, point.y, point.z + vz * seconds };
    return 1;
}
