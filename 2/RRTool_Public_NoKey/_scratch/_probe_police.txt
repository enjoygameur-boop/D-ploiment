### PICK
rr_aim_pick(const RR_PLAYER* players, int count, const RR_CAMERA* camera,
                       int width, int height, const BR_Config* config,
                       uint64_t locked_addr, int32_t locked_id, int (*clear)(RR_V3, RR_V3)) {
    RR_AIM_PICK best = {0}, held = {0};
    float best_score = INFINITY, held_ang = INFINITY;
    float cx = width * 0.5f, cy = height * 0.5f;
    float radius = (float)(config->aa_fov_px > 0 ? config->aa_fov_px : 150);
    /* The cone is derived from the drawn radius at the current zoom, so the ring
     * on screen and the limit used here describe the same angle. */
    float tan_half = tanf(camera->fov * 3.14159265f / 360.0f);
    float limit = (tan_half > 0.01f && height > 0)
                ? atanf(radius / ((height * 0.5f) / tan_half)) * 57.295779513f : 6.0f;
    float margin = (float)config->aa_switch_margin;
    int group = config->aa_bone;
    for (int i = 0; i < count; ++i) {
        const RR_PLAYER* p = &players[i];
        int is_held = locked_addr && p->addr == locked_addr && p->id == locked_id;
        if (!p->addr || !p->valid || p->dead || p->knocked || p->civil || p->is_vehicle || p->is_loot || p->local || p->hidden_in_object) continue;
        if (!rr_relationship_allowed(p->relationships, config->aa_target_clan, config->aa_target_police)) continue;
        if (!isfinite(p->dist) || (config->aa_dist > 0 && p->dist > (float)config->aa_dist)) continue;

        float sx, sy;
        RR_V3 point;
        if (!rr_aim_point(p, group, camera, width, height, &point) ||
            !game_w2s(camera, &point, width, height, &sx, &sy)) continue;
        float ax = point.x - camera->pos.x, ay = point.y - camera->pos.y, az = point.z - camera->pos.z;
        float len = sqrtf(ax*ax + ay*ay + az*az);
        if (!isfinite(len) || len < 0.05f) continue;
        float dot = (ax*camera->fwd.x + ay*camera->fwd.y + az*camera->fwd.z) / len;
        if (dot > 1.0f) dot = 1.0f; else if (dot < -1.0f) dot = -1.0f;
        float ang = acosf(dot) * 57.295779513f;
        if (!isfinite(ang)) continue;
        if (!is_held && ang > limit) continue;
        float dx = sx - cx, dy = sy - cy;
        if (!isfinite(dx) || !isfinite(dy)) continue;
        /* When the chosen joint is blocked, retry the same area for a clear one. */
        if (clear && !clear(camera->pos, point)) {
            if (!rr_aim_visible_point(p, group, camera, width, height, &point, clear) ||
                !game_w2s(camera, &point, width, height, &sx, &sy)) continue;
            ax = point.x - camera->pos.x; ay = point.y - camera->pos.y; az = point.z - camera->pos.z;
            len = sqrtf(ax*ax + ay*ay + az*az);
            if (!isfinite(len) || len < 0.05f) continue;
            dot = (ax*camera->fwd.x + ay*camera->fwd.y + az*camera->fwd.z) / len;
            if (dot > 1.0f) dot = 1.0f; else if (dot < -1.0f) dot = -1.0f;
            ang = acosf(dot) * 57.295779513f;
            if (!isfinite(ang)) continue;
            dx = sx - cx; dy = sy - cy;
            if (!isfinite(dx) || !isfinite(dy)) continue;
        }
        if (is_held) { held = (RR_AIM_PICK){p, point, dx, dy}; held_ang = ang; continue; }
        float score = ang + 0.02f * p->dist;
        if (score < best_score) { best = (RR_AIM_PICK){p, point, dx, dy}; best_score = score; }
    }
    /* A held lock keeps the pointer until a rival beats it by the switch margin. */
    if (held.player && (!best.player || best_score + margin >= held_ang)) return held;
    return best;
}
